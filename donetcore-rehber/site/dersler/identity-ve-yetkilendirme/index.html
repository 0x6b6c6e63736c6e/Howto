Identity ve veritabanı

Identity kullanıcıları, parola özetlerini, rolleri ve oturum bilgilerini yönetir. IdentityDbContext mevcut etkinlik tablolarının yanına Identity modelini ekler. base.OnModelCreating Identity eşlemelerini kurar. Örnekte IdentityEkle migration dosyası hazırdır.

Data/EtkinlikDbContext.cs
using Etkinlikler.Models;
using Microsoft.AspNetCore.Identity.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore;

namespace Etkinlikler.Data;

public sealed class EtkinlikDbContext(DbContextOptions<EtkinlikDbContext> options)
    : IdentityDbContext(options)
{
    public DbSet<Etkinlik> Etkinlikler => Set<Etkinlik>();

    public DbSet<EtkinlikKaydi> EtkinlikKayitlari => Set<EtkinlikKaydi>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);
        var etkinlik = modelBuilder.Entity<Etkinlik>();

        etkinlik.ToTable("Etkinlikler", table =>
            table.HasCheckConstraint("CK_Etkinlikler_Kontenjan", "[Kontenjan] BETWEEN 1 AND 500"));

        etkinlik.Property(item => item.Ad).HasMaxLength(80).IsRequired();

        etkinlik.HasData(
            new Etkinlik { Id = 1, Ad = "Web Geliştirme Atölyesi", Kontenjan = 25 },
            new Etkinlik { Id = 2, Ad = "SQL Server Semineri", Kontenjan = 40 });

        var katilimci = modelBuilder.Entity<Katilimci>();
        katilimci.ToTable("Katilimcilar");
        katilimci.Property(item => item.Ad).HasMaxLength(80).IsRequired();
        katilimci.HasData(
            new Katilimci { Id = 1, Ad = "Deniz Kaya" },
            new Katilimci { Id = 2, Ad = "Ece Arslan" },
            new Katilimci { Id = 3, Ad = "Mert Yılmaz" });

        var kayit = modelBuilder.Entity<EtkinlikKaydi>();
        kayit.ToTable("EtkinlikKayitlari");
        kayit.HasKey(item => new { item.EtkinlikId, item.KatilimciId });
        kayit.HasOne<Etkinlik>().WithMany().HasForeignKey(item => item.EtkinlikId);
        kayit.HasOne<Katilimci>().WithMany().HasForeignKey(item => item.KatilimciId);
        kayit.HasData(
            new EtkinlikKaydi { EtkinlikId = 1, KatilimciId = 1 },
            new EtkinlikKaydi { EtkinlikId = 1, KatilimciId = 2 },
            new EtkinlikKaydi { EtkinlikId = 1, KatilimciId = 3 });
    }
}

Identity ve oturum yapılandırması

AddIdentity kullanıcı ve rol servislerini, AddEntityFrameworkStores SQL Server veri erişimini kurar. Authentication oturum çerezini okur. Authorization rol kurallarını uygular. Sıralama önemlidir.

Program.cs
using Etkinlikler.Data;
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;

if (args is ["--rapor-sql"])
{
    // SQL üretmek için sağlayıcı gerekir. Bu bağlantı açılmaz.
    var options = new DbContextOptionsBuilder<EtkinlikDbContext>()
        .UseSqlServer("Server=localhost;Database=DotNetAtolyeEtkinlikler;Integrated Security=True;Encrypt=True")
        .Options;
    using var db = new EtkinlikDbContext(options);
    Console.WriteLine(DolulukRaporu.Sorgu(db).ToQueryString());
    return;
}

var yoneticiOlustur = args is ["--yonetici-olustur"];
var kayitDeneyi = args is ["--kayit-deneyi"];
var builder = WebApplication.CreateBuilder(kayitDeneyi || yoneticiOlustur ? [] : args);
builder.Services.AddControllersWithViews();
builder.Services.AddScoped<EtkinlikKayitServisi>();
builder.Services.AddIdentity<IdentityUser, IdentityRole>()
    .AddEntityFrameworkStores<EtkinlikDbContext>()
    .AddDefaultTokenProviders();
builder.Services.ConfigureApplicationCookie(options =>
{
    options.LoginPath = "/Hesap/Giris";
    options.AccessDeniedPath = "/Hesap/ErisimEngellendi";
});

var connection = builder.Configuration.GetConnectionString("EtkinlikDb")
    ?? throw new InvalidOperationException("ConnectionStrings:EtkinlikDb ayarını girin.");

builder.Services.AddDbContext<EtkinlikDbContext>(options =>
    options.UseSqlServer(connection));

await using var app = builder.Build();

if (yoneticiOlustur)
{
    await YoneticiKurulumu.CalistirAsync(app.Services, builder.Configuration);
    return;
}

if (kayitDeneyi)
{
    using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(30));
    await KayitDeneyi.CalistirAsync(app.Services, timeout.Token);
    return;
}

app.UseAuthentication();
app.UseAuthorization();

app.MapControllerRoute(
    name: "default",
    pattern: "{controller=Etkinlikler}/{action=Index}/{id?}");

await app.RunAsync();

Yönetim ekranlarını koru

[Authorize(Roles = "Yonetici")] rol denetimini sunucuda uygular. Etkinlik listeleme herkese açıktır. Etkinlik ekleme, doluluk raporu ve katılımcı kaydı yöneticiye açıktır. Menüde bağlantıyı gizlemek sunucu kontrolünü tamamlar.

Controllers/EtkinliklerController.cs
using Etkinlikler.Data;
using Microsoft.AspNetCore.Authorization;
using Etkinlikler.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace Etkinlikler.Controllers;

public class EtkinliklerController(EtkinlikDbContext db) : Controller
{
    [HttpGet]
    public async Task<IActionResult> Index(CancellationToken cancellationToken)
    {
        var etkinlikler = await db.Etkinlikler
            .AsNoTracking()
            .OrderBy(etkinlik => etkinlik.Id)
            .ToListAsync(cancellationToken);

        return View(etkinlikler);
    }

    [Authorize(Roles = "Yonetici")]
    [HttpGet]
    public async Task<IActionResult> Doluluk(CancellationToken cancellationToken)
    {
        var rapor = await DolulukRaporu.Sorgu(db).ToListAsync(cancellationToken);
        return View(rapor);
    }

    [Authorize(Roles = "Yonetici")]
    [HttpGet]
    public IActionResult Ekle() => View(new EtkinlikFormu());

    [Authorize(Roles = "Yonetici")]
    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Ekle(EtkinlikFormu form, CancellationToken cancellationToken)
    {
        if (!ModelState.IsValid)
        {
            return View(form);
        }

        db.Etkinlikler.Add(new Etkinlik
        {
            Ad = form.Ad!,
            Kontenjan = form.Kontenjan!.Value
        });

        await db.SaveChangesAsync(cancellationToken);
        return RedirectToAction(nameof(Index));
    }
}
Controllers/KayitlarController.cs
using Etkinlikler.Data;
using Microsoft.AspNetCore.Authorization;
using Etkinlikler.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Rendering;
using Microsoft.EntityFrameworkCore;

namespace Etkinlikler.Controllers;

[Authorize(Roles = "Yonetici")]
public class KayitlarController(EtkinlikDbContext db, EtkinlikKayitServisi kayitServisi) : Controller
{
    [HttpGet]
    public Task<IActionResult> Ekle(CancellationToken cancellationToken)
        => FormuGosterAsync(new KatilimciKayitFormu(), cancellationToken);

    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Ekle(KatilimciKayitFormu form, CancellationToken cancellationToken)
    {
        if (!ModelState.IsValid)
        {
            return await FormuGosterAsync(form, cancellationToken);
        }

        var sonuc = await kayitServisi.KaydetAsync(
            form.EtkinlikId!.Value, form.KatilimciId!.Value, cancellationToken);

        if (sonuc == KayitSonucu.Basarili)
        {
            TempData["KayitMesaji"] = "Katılımcı kaydedildi.";
            return RedirectToAction("Doluluk", "Etkinlikler");
        }

        var mesaj = sonuc switch
        {
            KayitSonucu.EtkinlikBulunamadi => "Etkinlik bulunamadı.",
            KayitSonucu.KatilimciBulunamadi => "Katılımcı bulunamadı.",
            KayitSonucu.ZatenKayitli => "Bu katılımcı etkinliğe zaten kayıtlı.",
            _ => "Etkinliğin kontenjanı dolu."
        };

        ModelState.AddModelError(string.Empty, mesaj);
        return await FormuGosterAsync(form, cancellationToken);
    }

    private async Task<IActionResult> FormuGosterAsync(
        KatilimciKayitFormu form, CancellationToken cancellationToken)
    {
        var etkinlikler = await db.Etkinlikler.AsNoTracking().OrderBy(e => e.Ad).ThenBy(e => e.Id)
            .ToListAsync(cancellationToken);
        var katilimcilar = await db.Set<Katilimci>().AsNoTracking().OrderBy(k => k.Ad).ThenBy(k => k.Id)
            .ToListAsync(cancellationToken);

        ViewBag.Etkinlikler = new SelectList(etkinlikler, nameof(Etkinlik.Id), nameof(Etkinlik.Ad));
        ViewBag.Katilimcilar = new SelectList(katilimcilar, nameof(Katilimci.Id), nameof(Katilimci.Ad));
        return View(nameof(Ekle), form);
    }
}

Giriş ve çıkış

PasswordSignInAsync parolayı doğrular ve oturum çerezi oluşturur. Başarısız denemelerde hesap kilidi uygulanır. Yönlendirme adresi yerel adres olarak doğrulanır. Çıkış POST isteği ve doğrulama belirteciyle yapılır.

Controllers/HesapController.cs
using Etkinlikler.Models;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.AspNetCore.Mvc;

namespace Etkinlikler.Controllers;

public class HesapController(SignInManager<IdentityUser> giris) : Controller
{
    [HttpGet]
    public IActionResult Giris(string? returnUrl) => View(new GirisFormu { ReturnUrl = returnUrl });

    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Giris(GirisFormu form)
    {
        if (!ModelState.IsValid) return View(form);

        var sonuc = await giris.PasswordSignInAsync(form.Eposta!, form.Parola!, false, lockoutOnFailure: true);
        if (sonuc.Succeeded)
        {
            return LocalRedirect(Url.IsLocalUrl(form.ReturnUrl) ? form.ReturnUrl! : "/");
        }

        ModelState.AddModelError(string.Empty, sonuc.IsLockedOut
            ? "Hesap geçici olarak kilitlendi."
            : "E-posta veya parola geçersiz.");
        return View(form);
    }

    [Authorize]
    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Cikis()
    {
        await giris.SignOutAsync();
        return RedirectToAction("Index", "Etkinlikler");
    }

    [HttpGet]
    public IActionResult ErisimEngellendi() => View();
}

Veritabanını ve yöneticiyi hazırla

Örnek ZIP’i aç, Etkinlikler klasörüne gir ve üçüncü dersteki SQL Server bağlantısını ayarla. Migration Identity tablolarını ekler. Geliştirme ortamında kendi e-posta ve güçlü parolanı User Secrets ile gir. Komut, kullanıcı yoksa oluşturur ve Yonetici rolünü atar. Mevcut kullanıcı varsa parolasını değiştirmez.

PowerShell’de önce $env:ASPNETCORE_ENVIRONMENT = "Development" çalıştır ve komutlardan ortam değişkeni ön ekini çıkar. User Secrets geliştirme içindir. Sunucuda gizli değerleri güvenli yapılandırma üzerinden sağla.

Data/YoneticiKurulumu.cs
using Microsoft.AspNetCore.Identity;

namespace Etkinlikler.Data;

public static class YoneticiKurulumu
{
    public static async Task CalistirAsync(IServiceProvider services, IConfiguration config)
    {
        var email = config["IlkYonetici:Eposta"] ?? throw new InvalidOperationException("İlk yönetici e-postasını ayarlayın.");
        var parola = config["IlkYonetici:Parola"] ?? throw new InvalidOperationException("İlk yönetici parolasını ayarlayın.");
        await using var scope = services.CreateAsyncScope();
        var roller = scope.ServiceProvider.GetRequiredService<RoleManager<IdentityRole>>();
        var kullanicilar = scope.ServiceProvider.GetRequiredService<UserManager<IdentityUser>>();

        if (!await roller.RoleExistsAsync("Yonetici"))
        {
            var rolSonucu = await roller.CreateAsync(new IdentityRole("Yonetici"));
            if (!rolSonucu.Succeeded) throw new InvalidOperationException(string.Join(", ", rolSonucu.Errors.Select(e => e.Description)));
        }

        var kullanici = await kullanicilar.FindByEmailAsync(email);
        if (kullanici is null)
        {
            kullanici = new IdentityUser { UserName = email, Email = email };
            var sonuc = await kullanicilar.CreateAsync(kullanici, parola);
            if (!sonuc.Succeeded) throw new InvalidOperationException(string.Join(", ", sonuc.Errors.Select(e => e.Description)));
        }

        if (!await kullanicilar.IsInRoleAsync(kullanici, "Yonetici"))
        {
            var sonuc = await kullanicilar.AddToRoleAsync(kullanici, "Yonetici");
            if (!sonuc.Succeeded) throw new InvalidOperationException(string.Join(", ", sonuc.Errors.Select(e => e.Description)));
        }
        Console.WriteLine("Yönetici hesabı ve rolü hazır.");
    }
}

Giriş ve yetkiyi kontrol et

http://localhost:5085/ adresinde etkinlik listesi açılır. Oturum açmadan /Etkinlikler/Ekle veya /Kayitlar/Ekle adresine gitmek giriş ekranına yönlendirir. Yönetici hesabıyla giriş yaptıktan sonra yönetim bağlantıları açılır. Çıkış yaptıktan sonra aynı adresler yeniden giriş ister.

Rolü olmayan bir hesapla oturum açıldığında yönetim ekranı erişim engellendi sayfasına yönlendirir. Bu kontrol için veritabanında ayrı bir kullanıcı oluşturup bu kullanıcıya yönetici rolü atamadan deneyebilirsin.
